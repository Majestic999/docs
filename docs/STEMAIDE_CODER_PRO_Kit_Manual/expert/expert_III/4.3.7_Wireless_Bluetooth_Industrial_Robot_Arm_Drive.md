## PROGRAMMING

**Step 1:** Open your Arduino IDE. See how to set up here: [Getting Started](../../Getting Started/Arduino_IDE_Setup.md).

**Step 2:** Write the complete program implementing the system logic with appropriate pin definitions, setup configuration, and the main control loop.

```cpp
#include <ESP32Servo.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <SoftwareSerial.h>

// Initialize components
SoftwareSerial bt(2, 3); // RX, TX
LiquidCrystal_I2C lcd(0x27, 16, 2); 
Servo s1, s2, s3, s4;

void setup() {
  bt.begin(9600);
  lcd.init();
  lcd.backlight();
  
  s1.attach(5); s2.attach(6); s3.attach(9); s4.attach(10);
  
  lcd.print("Arm Ready!");
}

void loop() {
  if (bt.available()) {
    char cmd = bt.read();
    lcd.clear();
    lcd.setCursor(0,0);
    
    // Example: Move servos based on received character
    switch(cmd) {
      case '1': s1.write(90); lcd.print("Base: 90 deg"); break;
      case '2': s2.write(90); lcd.print("Arm: 90 deg"); break;
      case '3': s3.write(90); lcd.print("Wrist: 90 deg"); break;
      case '4': s4.write(90); lcd.print("Claw: 90 deg"); break;
      case '0': // Reset
        s1.write(0); s2.write(0); s3.write(0); s4.write(0);
        lcd.print("Reset Position");
        break;
    }
  }
}
```
