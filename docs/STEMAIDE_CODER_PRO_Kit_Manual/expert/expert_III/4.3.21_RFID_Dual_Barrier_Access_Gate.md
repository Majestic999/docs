## PROGRAMMING

**Step 1:** Open your Arduino IDE. See how to set up here: [Getting Started](../../Getting Started/Arduino_IDE_Setup.md).

**Step 2:** Write the complete program implementing the system logic with appropriate pin definitions, setup configuration, and the main control loop.

```cpp
#include <SPI.h>
#include <MFRC522.h>
#include <ESP32Servo.h>

#define SS_PIN 33
#define RST_PIN 23
MFRC522 mfrc522(SS_PIN, RST_PIN);
Servo gateIn, gateOut;

const int red = 2, yellow = 3, green = 4;

void setup() {
  SPI.begin(); mfrc522.PCD_Init();
  gateIn.attach(5); gateOut.attach(6);
  pinMode(red, OUTPUT); pinMode(yellow, OUTPUT); pinMode(green, OUTPUT);
  
  // Set initial state
  gateIn.write(0); gateOut.write(0);
  digitalWrite(red, HIGH);
}

void loop() {
  if (!mfrc522.PICC_IsNewCardPresent() || !mfrc522.PICC_ReadCardSerial()) return;

  // Access Granted Logic
  digitalWrite(red, LOW); digitalWrite(green, HIGH);
  
  gateIn.write(90); gateOut.write(90); // Open barriers
  delay(3000); 
  
  gateIn.write(0); gateOut.write(0);   // Close barriers
  digitalWrite(green, LOW); digitalWrite(red, HIGH);
  
  mfrc522.PICC_HaltA();
}
```
