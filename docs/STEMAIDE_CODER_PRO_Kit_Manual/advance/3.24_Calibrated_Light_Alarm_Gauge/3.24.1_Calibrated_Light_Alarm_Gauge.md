**Step 2:** Connecting the potentiometer.

Connect the right pin to the GND rail.

Connect the middle (wiper) pin to Analog GPIO 36.

![Wiring diagram](../../../assets/3.0/Calibrated_Light_Alarm_Gauge/circuit_3.png)

**Step 2:** Connecting the LDR module.

Connect VCC to the 5V rail.

Connect the positive (+) pin of the buzzer to Digital GPIO 19.

## PROGRAMMING

**Step 1:** Open your Arduino IDE. See how to set up here: [Getting Started](../../Getting Started/Arduino_IDE_Setup.md).

**Step 2:** Write the complete program implementing the system logic with appropriate pin definitions, setup configuration, and the main control loop.

```cpp
// Pin Definitions
const int potPin = 36;
const int ldrPin = 39;

const int redLED = 4;
const int yellowLED = 5;
const int greenLED = 18;

const int buzzerPin = 19;

void setup()
{
  pinMode(redLED, OUTPUT);
  pinMode(yellowLED, OUTPUT);
  pinMode(greenLED, OUTPUT);

  pinMode(buzzerPin, OUTPUT);

  Serial.begin(9600);
}

void loop()
{
  int threshold = analogRead(potPin);
  int lightLevel = analogRead(ldrPin);

  Serial.print("Threshold: ");
  Serial.print(threshold);

  Serial.print("  Light: ");
  Serial.println(lightLevel);

  if (lightLevel < threshold - 50)
  {
    // Safe
    digitalWrite(greenLED, HIGH);
    digitalWrite(yellowLED, LOW);
    digitalWrite(redLED, LOW);
    noTone(buzzerPin);
  }
  else if (lightLevel >= threshold - 50 && lightLevel <= threshold)
  {
    // Warning
    digitalWrite(greenLED, LOW);
    digitalWrite(yellowLED, HIGH);
    digitalWrite(redLED, LOW);
    noTone(buzzerPin);
  }
  else
  {
    // Alarm
    digitalWrite(greenLED, LOW);
    digitalWrite(yellowLED, LOW);
    digitalWrite(redLED, HIGH);

    tone(buzzerPin, 1000);
  }

  delay(100);
}
```
